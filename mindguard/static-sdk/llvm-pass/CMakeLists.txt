cmake_minimum_required(VERSION 3.20)
project(mindguard_llvm_pass LANGUAGES CXX)

find_package(LLVM REQUIRED CONFIG)
if(NOT LLVM_PACKAGE_VERSION VERSION_GREATER_EQUAL 18 OR LLVM_PACKAGE_VERSION VERSION_GREATER_EQUAL 20)
  message(FATAL_ERROR "MindGuardPass requires LLVM 18.x or 19.x, found ${LLVM_PACKAGE_VERSION}")
endif()
add_library(MindGuardPass MODULE src/MindGuardPass.cpp)
target_compile_features(MindGuardPass PRIVATE cxx_std_20)
target_include_directories(MindGuardPass SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_compile_definitions(MindGuardPass PRIVATE ${LLVM_DEFINITIONS})
target_compile_options(MindGuardPass PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_link_libraries(MindGuardPass PRIVATE LLVM)
set_target_properties(MindGuardPass PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
